\chapter*{Wykaz skrótów}
\addcontentsline{toc}{chapter}{Wykaz skrótów}

\begin{description}
  \item[AE] autokoder (autoencoder)
  \item[API] interfejs programowania aplikacji (application programming interface)
  \item[B³] metryka Bagga--Baldwin $B^3$
  \item[BLANC] miara decyzji koreferencyjnych i niekoreferencyjnych
  \item[CEAF$_e$] entity-based Constrained Entity-Alignment F-measure
  \item[CoNLL F1] średnia F1 metryk MUC, B³ i CEAF$_e$
  \item[CPU] centralna jednostka przetwarzająca (central processing unit)
  \item[DAE] odszumiający autokoder (denoising autoencoder)
  \item[ELI] European Legislation Identifier
  \item[GPU] procesor graficzny (graphics processing unit)
  \item[JSON] JavaScript Object Notation
  \item[LEA] Link-based Entity-Aware score
  \item[LLM] duży model językowy (large language model)
  \item[MUC] Message Understanding Conference; metryka linkowa koreferencji
  \item[NLP] przetwarzanie języka naturalnego (natural language processing)
  \item[OCR] optyczne rozpoznawanie znaków (optical character recognition)
  \item[PCC] Polish Coreference Corpus
  \item[PII] dane umożliwiające identyfikację osoby (personally identifiable information)
  \item[t-SNE] t-distributed stochastic neighbor embedding
  \item[U-Net] konwolucyjna architektura enkoder--dekoder z połączeniami skrótowymi
  \item[VAE] wariacyjny autokoder (variational autoencoder)
  \item[YAML] YAML Ain't Markup Language
\end{description}
